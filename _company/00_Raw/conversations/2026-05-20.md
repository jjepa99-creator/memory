// 💡 스크롤 깊이 추적을 위한 IntersectionObserver
export const useScrollDepthTracker = (elementRef: React.RefObject<HTMLElement>) => {
    useEffect(() => {
        const element = elementRef.current;
        if (!element) return;

        const observerOptions = {
            root: null, // 뷰포트 기준
            rootMargin: '0px',
            threshold: 0.2 // 요소의 20%가 보일 때 트리거
        };

        const observer = new IntersectionObserver((entries) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    // 사용자가 이 섹션에 진입했음을 감지
                    trackEvent('scroll_depth_reached', { 
                        section_name: entry.target.dataset.section || 'unknown',
                        depth_percentage: '20%' // 실제 로직에서는 더 정교한 계산 필요
                    });
                    // 한 번 감지되면 더 이상 감지하지 않도록 관찰 중지
                    observer.unobserve(entry.target);
                }
            });
        }, observerOptions);

        observer.observe(element);

        return () => {
            observer.disconnect();
        };
    }, [elementRef]);
};

// 💡 클릭 이벤트 추적을 위한 훅
export const useClickTracker = (elementRef: React.RefObject<HTMLElement>, eventName: string) => {
    useEffect(() => {
        const element = elementRef.current;
        if (!element) return;

        const handleClick = (e: MouseEvent) => {
            // 기본 동작 방지 (링크 이동 등)
            // e.preventDefault(); 
            
            // 이벤트 추적 호출
            trackEvent(eventName, { 
                element_id: element.id || 'N/A',
                element_text: element.textContent?.trim() || 'N/A',
                page_location: window.location.pathname
            });
        };

        element.addEventListener('click', handleClick);

        return () => {
            element.removeEventListener('click', handleClick);
        };
    }, [elementRef, eventName]);
};

### 🎯 다음 목표: MVP 정의 및 핵심 성공 지표(KPI) 기반의 검증 프레임워크 구축

**[세부 실행 계획 및 담당자]**

#### 1. MVP 범위 확정 및 우선순위 재설정 (Product/Strategy)
*   **내용:** 현재까지 정의된 모든 기능 목록 중, **가장 적은 리소스로 가장 큰 가치를 검증할 수 있는 최소한의 핵심 기능(Core Loop)만 남깁니다.**
*   **산출물:** MVP 기능 목록 및 기능별 개발/검증 우선순위(Must-Have > Should-Have > Nice-to-Have)가 포함된 로드맵 초안.
*   **담당:** 기획/전략팀 (본인)

#### 2. 핵심 성공 지표(KPI) 및 지표 측정 로직 설계 (Data/Analytics)
*   **내용:** 우리 서비스의 성공을 정의할 핵심 지표(KPI) 3~5개를 확정합니다. (예: 전환율(Conversion Rate), 리텐션(Retention Rate), 사용자당 평균 세션 수(DAU/MAU)). 각 KPI를 측정하기 위한 **데이터 수집 로직(Tracking Logic)**을 구체적으로 설계합니다.
*   **산출물:** KPI 정의서 및 데이터 플로우 다이어그램 (어떤 행동 $\rightarrow$ 어떤 데이터 포인트 $\rightarrow$ 어떤 지표에 기여).
*   **담당:** 데이터 분석가 (또는 본인)

#### 3. A/B 테스트 설계 및 실행 계획 수립 (Testing)
*   **내용:** 가장 높은 불확실성을 가진 가설(예: '사용자는 이 버튼을 누르면 구매할 것이다')에 대해 A/B 테스트를 설계합니다. 테스트할 변수(Variable), 테스트할 그룹(A/B), 그리고 성공으로 간주할 기준(Success Metric)을 명확히 합니다.
*   **산출물:** A/B 테스트 계획서 (가설 $\rightarrow$ 변수 $\rightarrow$ 검증 기간 $\rightarrow$ 기대 효과).
*   **담당:** 마케팅/UX 리서치팀 (또는 본인)
