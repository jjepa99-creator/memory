```python
class BollingerBandsStrategy(BollingerBandsStrategy): # 상속 구조가 불필요하지만 명확성을 위해 유지
    def __init__(self, period=20, num_std=2):
        super().__init__()
        self.period = period
        self.num_std = num_std

    def generate_signals(self, data: pd.DataFrame) -> pd.Series:
        # 1. 이동평균 계산 (SMA)
        data['SMA'] = data['Close'].rolling(window=self.period).mean()
        # 2. 표준편차 계산 (STD)
        data['STD'] = data['Close'].rolling(window=self.period).std()
        # 3. 상한/하한 밴드 계산
        data['Upper_Band'] = data['SMA'] + (self.num_std * data['STD'])
        data['Lower_Band'] = data['SMA'] - (self.num_std * data['STD'])

        # 매수 시그널: 종가가 하한 밴드 근처일 때 (1)
        buy_signal = (data['Close'] < data['Lower_Band']).astype(int)
        # 매도 시그널: 종가가 상한 밴드 근처일 때 (1)
        sell_signal = (data['Close'] > data['Upper_Band']).astype(int)
        
        # 시그널 조합 (Buy=1, Sell=-1, Hold=0)
        signals = pd.Series(0, index=data.index)
        signals[buy_signal == 1] = 1
        signals[sell_signal == 1] = -1
        return signals.fillna(0)

# -------------------------------------------------------------------
# 2. RSI 기반 매매 (Relative Strength Index)
# 매수: RSI가 30 이하일 때 (과매도)
# 매도: RSI가 70 이상일 때 (과매수)
# -------------------------------------------------------------------
class RSIStrategy:
    def __init__(self, period=14, overbought=70, oversold=30):
        self.period = period
        self.overbought = overbought
        self.oversold = oversold

    def generate_signals(self, data: pd.DataFrame) -> pd.Series:
        # 실제 RSI 계산 로직은 복잡하므로, 여기서는 가상의 RSI 계산을 가정합니다.
        # 실제 환경에서는 TA-Lib 같은 라이브러리가 필요합니다.
        # 임시로 RSI 값을 생성하고 로직을 구현합니다.
        data['RSI'] = 100 - (100 / (1 + np.exp(-0.005 * np.arange(len(data)) + 0.5))) # 가상 RSI
        
        # 매수 시그널: RSI <= oversold (1)
        buy_signal = (data['RSI'] <= self.oversold).astype(int)
        # 매도 시그널: RSI >= overbought (1)
        sell_signal = (data['RSI'] >= self.overbought).astype(int)
        
        signals = pd.Series(0, index=data.index)
        signals[buy_signal == 1] = 1
        signals[sell_signal == 1] = -1
        return signals.fillna(0)

# 3. 전략 실행 및 시뮬레이션
# -------------------------------------------------------------------
def backtest_strategy(data: pd.DataFrame, strategy_class):
    """주어진 데이터와 전략 클래스를 사용하여 백테스팅을 수행하고 수익률을 반환합니다."""
    print(f"\n--- 백테스팅 시작: {strategy_class.__name__} ---")
    
    # 시그널 생성
    if strategy_class == BollingerBandsStrategy:
        signals = strategy_class().generate_signals(data.copy())
    elif strategy_class == RSIStrategy:
        signals = strategy_class().generate_signals(data.copy())
    else:
        raise ValueError("지원하지 않는 전략 클래스입니다.")
    
    # 시뮬레이션 로직: 보유 여부 추적
    position = 0  # 0: 미보유, 1: 보유
    returns = []
    
    for i in range(1, len(data)):
        current_signal = signals.iloc[i]
        
        if current_signal == 1 and position == 0: # 매수 신호 발생 및 미보유 시
            position = 1
        elif current_signal == -1 and position == 1: # 매도 신호 발생 및 보유 시
            position = 0
        
        # 수익률 계산: 현재 보유 상태에 따른 수익률만 기록
        # (가장 단순화된 예시: 다음 날의 수익률에만 의존)
        daily_return = data['Close'].pct_change().iloc[i]
        
        if position == 1:
            returns.append(daily_return)
        else:
            returns.append(0.0)
            
    # 최종 수익률 계산
    cumulative_return = np.cumprod([1 + r for r in returns]).iloc[-1]
    print(f"총 누적 수익률 (백테스트 기간): {cumulative_return:.2f}배")
    return cumulative_return
```

**가정:** 필요한 의존성 파일(`data_collector.py`, `indicator.py`)은 이미 존재합니다.

**작업 목표:** `Strategy` 클래스 정의 및 초기화 로직 작성.

**예시 코드 (Python):**
```python
from abc import ABC, abstractmethod
# from data_collector import DataCollector # 실제 환경에서 임포트 될 모듈
# from indicator import Indicator # 실제 환경에서 임포트 될 모듈

class TradingStrategy(ABC):
    """
    트레이딩 전략의 추상 기본 클래스 (Abstract Base Class).
    모든 구체적인 전략은 이 인터페이스를 상속받아야 합니다.
    """
    def __init__(self, symbol: str, timeframe: str):
        self.symbol = symbol
        self.timeframe = timeframe
        print(f"Strategy Initialized for {symbol} on {timeframe} timeframe.")

    @abstractmethod
    def calculate_signal(self, historical_data: list) -> str:
        """
        주어진 히스토리컬 데이터 기반으로 매매 신호를 계산합니다.
        반환 값: 'BUY', 'SELL', 'HOLD' 중 하나여야 합니다.
        """
        pass

    @abstractmethod
    def execute_trade(self, signal: str, current_price: float) -> dict:
        """
        실제 거래 실행을 시뮬레이션하고 결과를 반환합니다.
        """
        pass

class MACDStrategy(TradingStrategy):
    """
    MACD 지표를 활용하는 구체적인 트레이딩 전략 구현체.
    """
    def __init__(self, symbol: str, timeframe: str, fast_period: int = 12, slow_period: int = 26, signal_period: int = 9):
        super().__init__(symbol, timeframe)
        self.fast_period = fast_period
        self.slow_period = slow_period
        self.signal_period = signal_period

    def calculate_signal(self, historical_data: list) -> str:
        """
        데이터를 받아 MACD 크로스오버를 감지하여 신호를 결정합니다.
        (실제로는 Indicator 클래스를 활용하여 MACD 값을 계산해야 함)
        """
        # --- 로직 시뮬레이션 ---
        if len(historical_data) < self.slow_period:
            return "HOLD" # 데이터 부족
        
        # 예시: MACD 값이 시그널 라인을 상향 돌파하면 BUY
        # (실제 구현에서는 indicator.calculate_macd(historical_data)를 호출)
        mock_macd_value = 0.01 # 임시 값
        mock_signal_value = 0.00 # 임시 값
        
        if mock_macd_value > mock_signal_value and mock_macd_value > 0:
            print("🚨 [MACD Signal Detected]: Golden Cross (BUY)")
            return "BUY"
        elif mock_macd_value < mock_signal_value and mock_macd_value < 0:
            print("🔻 [MACD Signal Detected]: Death Cross (SELL)")
            return "SELL"
        else:
            return "HOLD"

    def execute_trade(self, signal: str, current_price: float) -> dict:
        """
        매매 신호에 따른 가상의 거래 실행 결과 반환.
        """
        if signal == "BUY":
            return {"status": "SUCCESS", "action": "BUY", "price": current_price, "message": "매수 주문 실행 완료"}
        elif signal == "SELL":
            return {"status": "SUCCESS", "action": "SELL", "price": current_price, "message": "매도 주문 실행 완료"}
        else:
            return {"status": "HOLD", "action": "NONE", "price": current_price, "message": "유지 관망"}

# --- 사용 예시 ---
if __name__ == "__main__":
    # 가상의 데이터 셋 (실제로는 DataCollector가 채워 넣음)
    mock_data = [100, 101, 102, 103, 104, 105] 
    current_price = 106.0

    # 1. 전략 객체 생성
    macd_strategy = MACDStrategy(symbol="BTC/USD", timeframe="1H")

    # 2. 신호 계산
    signal = macd_strategy.calculate_signal(mock_data)
    print(f"\n>>> 계산된 최종 신호: {signal}")

    # 3. 거래 실행 시뮬레이션
    trade_result = macd_strategy.execute_trade(signal, current_price)
    print("\n--- 거래 실행 결과 ---")
    import json
    print(json.dumps(trade_result, indent=4))
```
